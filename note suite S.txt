DEF FACTORIELLE

def factorielle(n):
    f=1      #initialisation
    for a in range(1,n+1):
        f=f*a
    return f

def compter_voyelles(s):
    a="a"
    e="e"
    i="i"
    o="o"
    u="u"
    nb_voyelles=0
    for k in s:
        if k==a or k==e or k==i or k==o or k==u:
            nb_voyelles+=1
    return nb_voyelles

NB PREMIER OU PAS

def est_premier(n):
    if n<=1:
        return False
    nb=0
    for i in range(1,n+1):
        if n%i==0:
            nb+=1
    if nb==2:
        return True
    else:
        return False
est_premier(7)
est_premier(8)

DETECTEUR DE DIVISEUR COMMUN

def pgcd(a, b):
    r=a%b
    while r!=0:
        a=b
        b=r
        r=a%b
    return b 
pgcd(48, 18)  #6
pgcd(96, 81)   #3

def compter_occurrences(liste, valeur):

NB LE PLUS GRAND

def max_liste(liste):
    max=[]
    n=0
    for k in range(1,len(liste)):
        if liste[k]<liste[n]:
            max=[]
            max.append(liste[n])
        n=k
    return max
max_liste([3, 7, 2, 9, 4])
max_liste([-1, -5, -3])

def filtrer_pairs(liste):
    nouvelle=[]
    for k in range(len(liste)):
        if liste[k]%2==0:
            nouvelle.append(liste[k])

SI LA CHAINE SE LIT PAREIL DANS LES 2 SENS PAS TERMINE

def est_palindrome(s):
    l=""
    for k in s:
        l=k+l
    if s=l:
        return True
    else:
        return False

SUITE DE FIBONACCI

def fibonacci(n):
    a=1
    b=1
    for i in range(n-2):
        b=b+a
        a=b-a
    return b
print(fibonacci(9))
print(fibonacci(7))
print(fibonacci(13))

TRIE LA LISTE SANS .SORT

def tri_selection(liste):
    mini=[]
    for i in range(len(liste)):
        mini.append(min(liste))
        liste.remove(min(liste))
    return mini
tri_selection([5, 2, 9, 1, 3])
tri_selection([4, 4, 2])
